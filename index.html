from fastapi import FastAPI, Request from fastapi.middleware.cors import CORSMiddleware from sqlalchemy import create_engine, Column, Integer, String, Float from sqlalchemy.orm import declarative_base, sessionmaker import json import random app = FastAPI() # CORS (Mini App) app.add_middleware( CORSMiddleware, allow_origins=["*"], allow_credentials=True, allow_methods=["*"], allow_headers=["*"], ) # DB engine = create_engine("sqlite:///game.db", connect_args={"check_same_thread": False}) SessionLocal = sessionmaker(bind=engine) Base = declarative_base() class User(Base): __tablename__ = "users" id = Column(Integer, primary_key=True) telegram_id = Column(String, unique=True) balance = Column(Float, default=10) Base.metadata.create_all(bind=engine) # CASE DATA CASE_PRICE = 5 REWARDS = [1, 2, 5, 10, 20] # AUTH (Telegram Mini App) @app.post("/auth") async def auth(request: Request): data = await request.json() tg_user = json.loads(data["user"]) tg_id = str(tg_user["id"]) db = SessionLocal() user = db.query(User).filter(User.telegram_id == tg_id).first() if not user: user = User(telegram_id=tg_id, balance=10) db.add(user) db.commit() db.refresh(user) return { "telegram_id": user.telegram_id, "balance": user.balance } # CASE INFO @app.get("/case") def case(): return { "price": CASE_PRICE, "rewards": REWARDS } # OPEN CASE @app.post("/open/{telegram_id}") def open_case(telegram_id: str): db = SessionLocal() user = db.query(User).filter(User.telegram_id == telegram_id).first() if not user: return {"error": "user not found"} if user.balance < CASE_PRICE: return {"error": "not enough balance"} user.balance -= CASE_PRICE reward = random.choice(REWARDS) user.balance += reward db.commit() return { "reward": reward, "balance": user.balance } #
