// Рандомайзер для Gacha Case (TON экосистема)
type RewardType = 'ton' | 'item' | 'gift';

const TON_TIERS = {
    1: { min: 0.1, mid: 0.2, max: 0.5 },
    5: { min: 0.5, mid: 1.0, max: 2.5 },
    10: { min: 1.0, mid: 2.0, max: 5.0 },
    15: { min: 1.5, mid: 3.0, max: 7.5 },
    20: { min: 2.0, mid: 4.0, max: 10.0 }
};

function getRandomTonAmount(casePrice: number): number {
    const tier = TON_TIERS[casePrice];
    const rand = Math.random();
    // веса: 50% min, 30% mid, 20% max
    if (rand < 0.5) return tier.min;
    if (rand < 0.8) return tier.mid;
    return tier.max;
}

function openCase(casePrice: number): { type: RewardType; value: string | number; details: string } {
    const r = Math.random(); // 0..1
    if (r < 0.7) { // 70% TON дроп
        const amount = getRandomTonAmount(casePrice);
        return { type: 'ton', value: amount, details: `Выигрыш ${amount} TON` };
    } else if (r < 0.9) { // 20% внутриигровой предмет
        const items = ['⭐ Кибер-карта "Фантом"', '🔫 Скин "Неоновый дракон"', '💠 Осколок Кванта', '🌀 Эффект Вортекса', '⚡ Молния Хаоса'];
        const randomItem = items[Math.floor(Math.random() * items.length)];
        return { type: 'item', value: randomItem, details: `Предмет: ${randomItem}` };
    } else { // 10% NFT Gift (Telegram Gift)
        const gifts = ['🎁 Neon Rose Gift', '🎁 TON Crystal Gift', '🎁 Cyber Star Gift', '🎁 Quantum Badge'];
        const gift = gifts[Math.floor(Math.random() * gifts.length)];
        return { type: 'gift', value: gift, details: `✨ Редкий NFT Gift: ${gift}` };
    }
}
Платежная система: TON & Telegram Gifts

Метод 1: Пополнение через Telegram Gifts — пользователь отправляет Telegram Gift боту-хранилищу @Gift_Storage_Bot. Webhook ловит событие, верифицирует отправителя и конвертирует подарок в TON по рыночной цене, зачисляя внутренний баланс.

Метод 2: Прямая TON транзакция — интеграция с TON Connect / Telegram Wallet. Пользователь подписывает платеж, после подтверждения бэкенд начисляет баланс.

Webhook-логика для Telegram Gifts (Node.js/Express пример): 
// Эндпоинт webhook для бота
app.post('/webhook/gift', async (req, res) => {
  const { message } = req.body;
  if (!message || !message.text) return res.sendStatus(200);
  // Проверка: сообщение содержит gift, отправленное на @Gift_Storage_Bot
  const giftMatch = message.text.match(/🎁|Gift|подарок/i);
  const userId = message.from.id;
  if (giftMatch && message.forward_from_bot?.username === 'Gift_Storage_Bot') {
     // 1. Определить рыночную стоимость подарка (запрос к внешнему API / фиксированная цена)
     const giftValueInTon = 2.5; // пример: стоимость подарка ~2.5 TON
     // 2. Зачисляем на баланс пользователя в БД
     await db.addBalance(userId, giftValueInTon);
     // 3. Отправляем подтверждение
     await bot.sendMessage(userId, `✅ Подарок принят! +${giftValueInTon} TON на баланс`);
  }
  res.sendStatus(200);
});

📌 При получении подарка на юзернейм бота, он автоматически зачисляет эквивалент TON на внутренний баланс пользователя. Каждый подарок верифицируется (уникальный ID, чтобы избежать дублей).

Структура внутреннего баланса: user_id, ton_balance, items[] (id предметов), gifts[] (id nft подарков).

Вывод средств и NFT

Пользователь на экране «Вывод» видит накопленный TON баланс и список выигранных NFT Gift-ов. Запрос на вывод обрабатывается автоматическим скриптом бота-хранилища:

  • Вывод TON: система отправляет транзакцию на кошелек пользователя (через TON Wallet API) с указанной суммой, списывая с внутреннего баланса.
  • Вывод NFT Gift: бот пересылает выбранный подарок на аккаунт пользователя в Telegram (прямой перевод Gift).
// Логика вывода NFT Gifts (бот-менеджер)
async function withdrawGift(userId, giftId) {
   const gift = await db.getUserGift(userId, giftId);
   if (!gift) throw new Error("Gift not found");
   // Отправка оригинального Telegram Gifts через Bot API
   await bot.sendGift(userId, gift.tokenId);
   await db.removeGift(userId, giftId);
   return { success: true };
}

Для максимального ощущения открытия кейса необходимо интегрировать следующие визуальные приемы (Three.js/WebGL):

User Flow & Системная архитектура

User Flow: Запуск Mini App → Авторизация через Telegram WebApp → Выбор кейса → Спиcание TON с баланса (или через пополнение) → 3D анимация → Показ награды → Обновление инвентаря → Возможность вывода.

Бэкенд компоненты: Node.js (Nest/Express) + PostgreSQL для пользователей, инвентаря, логов; Redis для кэша; интеграция с TON SDK (@ton/ton) для платежей; webhook сервер для Telegram Gifts; очередь на отправку выводов.

Безопасность: Проверка подписей ton-proof, антифрод система лимитов открытия кейсов, валидация каждого WebApp запроса через initData.
