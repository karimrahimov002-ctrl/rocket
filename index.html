const http = require("http"); const WebSocket = require("ws"); /* ================= SERVER ================= */ let clients = []; let players = {}; let round = { phase: "waiting", multiplier: 1, timer: 5 }; function broadcast(data){ clients.forEach(c=>{ if(c.readyState===1){ c.send(JSON.stringify(data)); } }); } /* ===== GAME ===== */ function startRound(){ round.phase="playing"; round.multiplier=1; Object.values(players).forEach(p=>{ p.status="playing"; }); let loop=setInterval(()=>{ round.multiplier *= 1.008; let crashChance = 0.01 + round.multiplier*0.008; if(Math.random(){ if(p.status==="playing") p.status="lost"; }); broadcast({ type:"crash", multiplier:round.multiplier, players:Object.values(players) }); round.phase="waiting"; round.timer=5; let t=setInterval(()=>{ round.timer--; broadcast({ type:"waiting", timer:round.timer, online:clients.length }); if(round.timer<=0){ clearInterval(t); startRound(); } },1000); } /* ================= HTTP (HTML) ================= */ const html = ` Risk Gift
Баланс: 0 TON
Онлайн: 0
...
🚀
📩 Отправьте подарок в Telegram: @riskgift_bot
После этого вам придут TON на баланс автоматически
`; /* ================= SERVER ================= */ const server = http.createServer((req,res)=>{ res.writeHead(200,{"Content-Type":"text/html"}); res.end(html); }); const wss = new WebSocket.Server({ server }); wss.on("connection",(ws)=>{ const id=Math.random().toString(36).slice(2,9); clients.push(ws); players[id]={ id, name:"User_"+id.slice(0,4), balance:1000, bet:0, status:"idle" }; ws.send(JSON.stringify({ type:"init", player:players[id], round, online:clients.length })); ws.on("message",(msg)=>{ const data=JSON.parse(msg); if(data.type==="bet"){ let p=players[id]; if(p.balance>=data.amount){ p.balance-=data.amount; p.bet=data.amount; p.status="playing"; } } if(data.type==="cashout"){ let p=players[id]; if(p.status==="playing"){ let win=Math.floor(p.bet*round.multiplier); p.balance+=win; p.status="cash"; } } }); ws.on("close",()=>{ clients=clients.filter(c=>c!==ws); delete players[id]; }); }); server.listen(3000,()=>{ console.log("http://localhost:3000"); }); startRound();
